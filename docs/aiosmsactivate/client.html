 58    async def __send_request(self, action: str, **kwargs):
 59        last_exception = None
 60
 61        for url in self._base_urls:
 62            try:
 63                url = self._accept_url if self._accept_url else url
 64                params = kwargs.get('params')
 65                if params:
 66                    kwargs.pop('params')
 67                async with aiohttp.ClientSession() as session:
 68                    async with session.request(
 69                        'POST',
 70                        url,
 71                        **kwargs,
 72                        params={
 73                            'api_key': self._api_key,
 74                            'action': action,
 75                            **(params if params else {})
 76                        }
 77                    ) as response:
 78                        response.raise_for_status()
 79                        logging.debug(response.real_url)
 80                        return await response.text()
 81            except Exception as e:
 82                last_exception = e

 84            self._accept_url = url
